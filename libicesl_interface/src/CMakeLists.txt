CMAKE_MINIMUM_REQUIRED(VERSION 2.6)
PROJECT(IceSL-Interface)

INCLUDE_DIRECTORIES (
	${CMAKE_CURRENT_SOURCE_DIR}
	${CMAKE_CURRENT_SOURCE_DIR}/../include
)

ADD_DEFINITIONS(-DASIO_STANDALONE )
ADD_DEFINITIONS(-DUSE_GLUX)
ADD_DEFINITIONS(-DLIBARCHIVE_STATIC)

#LINK_DIRECTORIES( ${CMAKE_CURRENT_SOURCE_DIR}/lib )

IF(ICESL_USE_CXX17 AND NOT EMSCRIPTEN)
  SET(CMAKE_CXX_STANDARD 17)
  SET(CMAKE_CXX_STANDARD_REQUIRED ON)
ELSE()
  SET(CMAKE_CXX_STANDARD 11)
  SET(CMAKE_CXX_STANDARD_REQUIRED ON)
ENDIF()

IF(ICESL_USE_PLUGINS AND NOT EMSCRIPTEN)
  INCLUDE(PluginFPIC)
ENDIF()

SET(ICESL_SOURCE
  DynamicLinker.h
  DynamicLinker.cpp
  IPlugin.h
  IPluginEnvironment.h
  Localization.h
  Localization.cpp
  EnumerableSettingsInterface.h
  IPath.h
  PathTypeBitField.h
  Path.h
  Path.cpp
  ILayer.h
  # Plugins
  IInfillerPlugin.h
  )

ADD_LIBRARY(icesl_interface ${ICESL_SOURCE})
SET_PROPERTY(TARGET icesl_interface APPEND PROPERTY
   INTERFACE_INCLUDE_DIRECTORIES
 			${CMAKE_CURRENT_SOURCE_DIR}
 			${CMAKE_CURRENT_SOURCE_DIR}/../include
)

IF(EMSCRIPTEN)
  TARGET_LINK_LIBRARIES(icesl_interface LibSL LibSL_gl)
ELSE(EMSCRIPTEN)
  TARGET_LINK_LIBRARIES(icesl_interface LibSL LibSL_gl4core)
ENDIF()

TARGET_LINK_LIBRARIES(icesl_interface
  libinterface
  clipper
)

IF(NOT EMSCRIPTEN)
IF(ICESL_USE_CXX17)
IF(${CMAKE_CXX_COMPILER_ID} STREQUAL "Clang")
  TARGET_LINK_LIBRARIES( icesl_interface
    stdc++fs
#      c++fs c++abi
  )
ENDIF()
ENDIF()
ENDIF()

SET_TARGET_PROPERTIES(icesl_interface PROPERTIES DEBUG_POSTFIX "-d")
