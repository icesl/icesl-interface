CMAKE_MINIMUM_REQUIRED(VERSION 3.10)
PROJECT(sampleMachineCodePlugin)

INCLUDE(PluginFPIC)

INCLUDE_DIRECTORIES(
	${CMAKE_CURRENT_SOURCE_DIR}
)

IF(EMSCRIPTEN)
  SET(LIBRARY_TYPE STATIC)
ELSE(EMSCRIPTEN)
  SET(LIBRARY_TYPE MODULE)
ENDIF(EMSCRIPTEN)

ADD_LIBRARY(sampleMachineCodePlugin ${LIBRARY_TYPE}
  SampleMachineCodePlugin.h
  SampleMachineCodePlugin.cpp
  sh_simplerender.fp
  sh_simplerender.vp
  sh_simplerender.h
)

AUTO_BIND_SHADERS(
  sh_simplerender
)

TARGET_LINK_LIBRARIES(sampleMachineCodePlugin
  LibSL
  icesl_interface
)

SET_PROPERTY(TARGET sampleMachineCodePlugin APPEND PROPERTY
   INTERFACE_INCLUDE_DIRECTORIES
 			${CMAKE_CURRENT_SOURCE_DIR}
)

IF(ICESL_PROJECT)
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins-d")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
ELSE()
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
ENDIF()

SET_TARGET_PROPERTIES(sampleMachineCodePlugin PROPERTIES DEBUG_POSTFIX "-d")
