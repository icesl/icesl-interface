CMAKE_MINIMUM_REQUIRED(VERSION 2.6)
PROJECT(sampleInfillerPlugin)

IF(ICESL_USE_CXX17 AND NOT EMSCRIPTEN)
  SET(CMAKE_CXX_STANDARD 17)
  SET(CMAKE_CXX_STANDARD_REQUIRED ON)
ELSE()
  SET(CMAKE_CXX_STANDARD 11)
  SET(CMAKE_CXX_STANDARD_REQUIRED ON)
ENDIF()

INCLUDE(PluginFPIC)

INCLUDE_DIRECTORIES(
	${CMAKE_CURRENT_SOURCE_DIR}
)

ADD_LIBRARY(sampleInfillerPlugin MODULE
  SampleInfillerPlugin.h
  SampleInfillerPlugin.cpp
)

IF(NOT EMSCRIPTEN)
IF(${CMAKE_CXX_COMPILER_ID} STREQUAL "Clang")
  TARGET_LINK_LIBRARIES(sampleInfillerPlugin
    stdc++fs
  )
ENDIF()
ENDIF()

TARGET_LINK_LIBRARIES(sampleInfillerPlugin
  LibSL
  icesl_interface
)

SET_PROPERTY(TARGET sampleInfillerPlugin APPEND PROPERTY
   INTERFACE_INCLUDE_DIRECTORIES
 			${CMAKE_CURRENT_SOURCE_DIR}
)

IF(ICESL_PROJECT)
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
IF (WIN32)
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
ELSE()
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins-d")
ENDIF()
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
ELSE()
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
ENDIF()

SET_TARGET_PROPERTIES(sampleInfillerPlugin PROPERTIES DEBUG_POSTFIX "-d")
