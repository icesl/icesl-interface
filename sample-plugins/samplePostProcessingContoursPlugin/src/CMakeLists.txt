CMAKE_MINIMUM_REQUIRED(VERSION 2.6)
PROJECT(samplePostProcessingContoursPlugin)

INCLUDE(PluginFPIC)

INCLUDE_DIRECTORIES(
	${CMAKE_CURRENT_SOURCE_DIR}
)

ADD_LIBRARY(samplePostProcessingContoursPlugin MODULE
  samplePostProcessingContoursPlugin.h
  samplePostProcessingContoursPlugin.cpp
)

TARGET_LINK_LIBRARIES(samplePostProcessingContoursPlugin
  LibSL
  icesl_interface
)

SET_PROPERTY(TARGET samplePostProcessingContoursPlugin APPEND PROPERTY
   INTERFACE_INCLUDE_DIRECTORIES
 			${CMAKE_CURRENT_SOURCE_DIR}
)

IF(ICESL_PROJECT)
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins-d")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../icesl-plugins")
ELSE()
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
SET_TARGET_PROPERTIES(samplePostProcessingContoursPlugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/../../../bin")
ENDIF()

SET_TARGET_PROPERTIES(samplePostProcessingPlugin PROPERTIES DEBUG_POSTFIX "-d")
